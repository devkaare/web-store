package views

import (
	"fmt"

	"github.com/devkaare/web-store/views/components"
)

type CartProp struct {
	UserID    int
	ProductID int
	Size      string
	Quantity  int
	Name      string
	Price     int
	ImagePath string
}

templ CartPage(cart []CartProp) {
	@components.Base() {
		for _, i := range cart {
			<p>{ i.Name } <strong>{ string(templ.SafeURL(fmt.Sprintf("%d", i.Price))) }</strong></p>
			<p>{ i.Size } { string(templ.SafeURL(fmt.Sprintf("%d", i.Quantity))) }</p>
		}
	}
}
